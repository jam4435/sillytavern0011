function sanitizePromptValue(value) {
  if (Array.isArray(value)) return value.map(item => sanitizePromptValue(item));
  if (!value || typeof value !== 'object') return value;
  return Object.fromEntries(
    Object.entries(value)
      .filter(([key]) => !key.startsWith('$'))
      .map(([key, child]) => [key, sanitizePromptValue(child)]),
  );
}
function toStrictJson(value) {
  const result = JSON.stringify(sanitizePromptValue(value));
  return result === undefined ? 'null' : result;
}
function getAptitudeLevel(value) {
  const num = Number(value);
  if (!Number.isFinite(num)) return '未知';
  if (num <= 4) return '低劣';
  if (num <= 8) return '普通';
  if (num <= 12) return '上佳';
  if (num <= 16) return '出众';
  return '绝顶';
}

  if (!attrs || typeof attrs !== 'object') return '';

}
function trimTrailingPunctuation(text) {
  return String(text).trim().replace(/[。！？；;，,、]+$/, '');
}

function formatNearbyRumorForPrompt(value) {
  const source = String(value ?? '').replace(/\s+/g, ' ').trim().replace(/\|/g, '／');
  const match = source.match(/^(.*?)\s*\[([^/\]]+)\/(.+)\]$/);
  if (!match) return `未知日期|未知地点|${source}`;
  const date = match[2].match(/\d+年\d+月\d+日/)?.[0] || '未知日期';
  return `${date}|${match[3].trim().replace(/\|/g, '／')}|${match[1].trim().replace(/\|/g, '／')}`;
}
function normalizeFullLocationPath(value) {
  if (typeof value !== 'string') return '';
  const segments = value.trim().replace(/[\\＞>›→]+/g, '/').split('/').map(part => part.trim());
  if ((segments.length !== 3 && segments.length !== 4) || segments.some(part => !part)) return '';
  return segments.join('/');
}
function getLocationScopePath(value) {
  const normalized = normalizeFullLocationPath(value);
  return normalized ? normalized.split('/').slice(0, 3).join('/') : '';
}
function isSameLocationScope(left, right) {
  const leftScope = getLocationScopePath(left);
  return Boolean(leftScope) && leftScope === getLocationScopePath(right);
}
function uniqueFullLocationPaths(value) {
  if (!Array.isArray(value)) return [];
  return [...new Set(value.map(normalizeFullLocationPath).filter(Boolean))];
}
