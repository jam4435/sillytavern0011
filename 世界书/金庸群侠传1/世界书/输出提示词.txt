// 去引号的紧凑JSON格式，节省token
function toCompact(obj) {
  if (obj === null || obj === undefined) return 'null';
  if (typeof obj === 'number' || typeof obj === 'boolean') return String(obj);
  if (typeof obj === 'string') return obj.includes(',') || obj.includes(':') || obj.includes('\n') ? `"${obj}"` : obj;
  if (Array.isArray(obj)) {
    if (obj.length === 0) return '[]';
    return '[' + obj.map(v => toCompact(v)).join(',') + ']';
  }
  const entries = Object.entries(obj)
    .filter(([k]) => k !== '$meta' && k !== '$template')
    .map(([k, v]) => k + ':' + toCompact(v));
  if (entries.length === 0) return '{}';
  return '{' + entries.join(',') + '}';
}

function getAptitudeLevel(value) {
  const num = Number(value);
  if (!Number.isFinite(num)) return '未知';
  if (num <= 4) return '低劣';
  if (num <= 8) return '普通';
  if (num <= 12) return '上佳';
  if (num <= 16) return '出众';
  return '绝顶';
}

function describeInitialAttributes(attrs) {
  if (!attrs || typeof attrs !== 'object') return '';
  const entries = Object.entries(attrs)
    .filter(([k]) => k !== '$meta' && k !== '$template')
    .map(([k, v]) => `${k}${v}（${getAptitudeLevel(v)}）`);
  if (entries.length === 0) return '';
  return '初始属性：' + entries.join('|');
}

function trimTrailingPunctuation(text) {
  return String(text).trim().replace(/[。！？；;，,、]+$/, '');
}

}
