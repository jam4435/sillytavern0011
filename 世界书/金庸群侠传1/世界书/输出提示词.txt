function sanitizePromptValue(value) {
  if (Array.isArray(value)) return value.map(item => sanitizePromptValue(item));
  if (!value || typeof value !== 'object') return value;
  return Object.fromEntries(
    Object.entries(value)
      .filter(([key]) => !key.startsWith('$'))
      .map(([key, child]) => [key, sanitizePromptValue(child)]),
  );
}
function compactPromptScalar(value) {
  if (value === null || value === undefined) return '无';
  if (typeof value === 'number' || typeof value === 'boolean') return String(value);
  const text = String(value).replace(/\s+/g, ' ').trim();
  return text || '无';
}
function compactPromptKey(value) {
  return String(value ?? '')
    .replace(/\s+/g, ' ')
    .trim()
    .replace(/:/g, '：') || '未命名';
}
function toPromptYaml(value, indent = 0) {
  const sanitized = sanitizePromptValue(value);
  const pad = ' '.repeat(indent);
  if (Array.isArray(sanitized)) {
    if (sanitized.length === 0) return pad + '[]';
    return sanitized.map(item => {
      if (item && typeof item === 'object') {
        const nested = toPromptYaml(item, indent + 2);
        return pad + '-\n' + nested;
      }
      return pad + '- ' + compactPromptScalar(item);
    }).join('\n');
  }
  if (!sanitized || typeof sanitized !== 'object') return pad + compactPromptScalar(sanitized);
  const entries = Object.entries(sanitized);
  if (entries.length === 0) return pad + '{}';
  return entries.map(([key, child]) => {
    const safeKey = compactPromptKey(key);
    if (child && typeof child === 'object') {
      return pad + safeKey + ':\n' + toPromptYaml(child, indent + 2);
    }
    return pad + safeKey + ': ' + compactPromptScalar(child);
  }).join('\n');
}
function getAptitudeLevel(value) {

  if (num <= 4) return '低劣';
  if (num <= 8) return '普通';
  if (num <= 12) return '上佳';
  if (num <= 16) return '出众';
  return '绝顶';
}
function describeInitialAttributes(attrs) {
  if (!attrs || typeof attrs !== 'object') return '';
  return Object.entries(attrs)
    .filter(([key]) => !key.startsWith('$'))

}
function trimTrailingPunctuation(text) {
  return String(text).trim().replace(/[。！？；;，,、]+$/, '');
}
function describeTalents(talents) {
  if (!talents || typeof talents !== 'object') return '';
  return Object.entries(talents)
    .filter(([key]) => !key.startsWith('$'))
    .map(([key, value]) => `${key}:${trimTrailingPunctuation(value)}`)

}
function formatWorldTime(time) {
  if (!time || typeof time !== 'object') return '';
  return `${compactPromptScalar(time.年)}年${compactPromptScalar(time.月)}月${compactPromptScalar(time.日)}日 ${compactPromptScalar(time.时)}时${compactPromptScalar(time.分)}分`;
}
function formatNearbyRumorForPrompt(value) {
  const source = String(value ?? '').replace(/\s+/g, ' ').trim().replace(/\|/g, '／');
  const match = source.match(/^(.*?)\s*\[([^/\]]+)\/(.+)\]$/);
  if (!match) return `未知日期|未知地点|${source}`;
  const date = match[2].match(/\d+年\d+月\d+日/)?.[0] || '未知日期';
  return `${date}|${match[3].trim().replace(/\|/g, '／')}|${match[1].trim().replace(/\|/g, '／')}`;
}
function normalizeFullLocationPath(value) {
  if (typeof value !== 'string') return '';
  const segments = value.trim().replace(/[\\＞>›→]+/g, '/').split('/').map(part => part.trim());
  if ((segments.length !== 3 && segments.length !== 4) || segments.some(part => !part)) return '';
  return segments.join('/');
}
function getLocationScopePath(value) {
  const normalized = normalizeFullLocationPath(value);
  return normalized ? normalized.split('/').slice(0, 3).join('/') : '';
}
function isSameLocationScope(left, right) {
  const leftScope = getLocationScopePath(left);
  return Boolean(leftScope) && leftScope === getLocationScopePath(right);
}
function uniqueFullLocationPaths(value) {
  if (!Array.isArray(value)) return [];
  return [...new Set(value.map(normalizeFullLocationPath).filter(Boolean))];
}
